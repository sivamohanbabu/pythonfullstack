JavaScript Full Notes (Basic → Advanced)

All concepts with code examples inside a single HTML page.

1. Variables

// var – function scoped
var name = "John";

// let – block scoped
let age = 20;

// const – cannot be changed
const pi = 3.14;

2. Data Types

// Primitive types
let a = "Hello";
let b = 42;
let c = true;
let d;            // undefined
let e = null;

// Reference types
let obj = {name: "Arshad", age: 20};
let arr = [10, 20, 30];

3. Operators

// Arithmetic
let x = 5 + 3;  

// Comparison
5 == "5";    // true
5 === "5";   // false

// Logical
true && false;

4. Conditionals

// if / else
let age = 18;
if (age >= 18) {
    console.log("Adult");
} else {
    console.log("Minor");
}

// switch
let day = 2;
switch(day) {
    case 1: console.log("Monday"); break;
    case 2: console.log("Tuesday"); break;
    default: console.log("Other");
}

5. Loops

// for loop
for (let i = 1; i <= 5; i++) {
    console.log(i);
}

// while loop
let x = 1;
while(x <= 3) {
    console.log(x);
    x++;
}

// for...of
for (let n of [10,20,30]) {
    console.log(n);
}

// for...in
let person = {name:"Ava", age:22};
for (let key in person) {
    console.log(key, person[key]);
}

6. Functions

// Normal function
function add(a, b) {
    return a + b;
}

// Arrow function
const multiply = (a, b) => a * b;

// Function expression
const greet = function() {
    console.log("Hello");
};

7. Arrays

let nums = [1,2,3];

nums.push(4);   // add to end
nums.pop();     // remove end

nums.shift();   // remove first
nums.unshift(0); // add first

nums.map(n => n * 2);
nums.filter(n => n > 2);

nums.reduce((a,b) => a + b, 0);

8. Objects

let person = {
    name: "Ravi",
    age: 25,
    greet() {
        console.log("Hi");
    }
};

console.log(person.name);
person.greet();

9. DOM Manipulation

Try clicking the button below:

// Select elements
document.getElementById("btn");
document.querySelector(".class");

// Change content
document.getElementById("message").innerText = "Hello User";

// Events
document.getElementById("btn").addEventListener("click", () => {
    document.getElementById("message").innerText = "Button Clicked!";
});

10. Prototypes

function Person(name) {
    this.name = name;
}

Person.prototype.sayHi = function() {
    console.log("Hi " + this.name);
};

let p = new Person("Arshad");
p.sayHi();

11. Classes (ES6)

class Car {
    constructor(name) {
        this.name = name;
    }
    start() {
        console.log(this.name + " started");
    }
}

let c = new Car("BMW");
c.start();

12. Closures

function outer() {
    let count = 0;
    return function() {
        count++;
        console.log(count);
    };
}

let inc = outer();
inc(); // 1
inc(); // 2

13. Promises

const promise = new Promise((resolve, reject) => {
    setTimeout(() => resolve("Done"), 1000);
});

promise.then(result => console.log(result));

14. Async / Await

async function getData() {
    let response = await fetch("https://jsonplaceholder.typicode.com/todos/1");
    let data = await response.json();
    console.log(data);
}

getData();

15. Spread & Rest

// Spread
let nums1 = [1,2,3];
let nums2 = [...nums1, 4,5];

// Rest
function sum(...numbers) {
    return numbers.reduce((a,b)=>a+b);
}

16. Destructuring

let user = {name:"Arjun", age:23};
let {name, age} = user;

let arr = [10,20,30];
let [a, b, c] = arr;

17. LocalStorage

localStorage.setItem("name", "Arshad");
localStorage.getItem("name");
