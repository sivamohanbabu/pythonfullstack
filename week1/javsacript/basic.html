Variables

Declare values with var, let or const.

// var — function scoped (avoid modern use)
var x = 5;

// let — block scoped
let name = "Rama";

// const — block scoped, cannot be reassigned
const PI = 3.14159;

Loops

Repeat tasks — common loops below.

// for
for (let i=0;i<3;i++){ console.log(i) }

// while
let i=0;
while(i<3){ console.log(i); i++; }

// for...of (arrays)
for (const v of [10,20,30]) { console.log(v) }

// for...in (object keys)
for (const k in {a:1,b:2}) { console.log(k) }

Conditionals

// if / else
let age = 18;
if (age >= 18) { console.log('Adult') } else { console.log('Minor') }

// switch
let day = 2;
switch(day){
  case 1: console.log('Mon'); break;
  case 2: console.log('Tue'); break;
  default: console.log('Other');
}

Arrays

Ordered collections — many useful methods.

let nums = [1,2,3,4];

// common methods
nums.push(5);             // add
nums.pop();               // remove last
nums.map(n => n * 2);    // create new array
nums.filter(n => n > 2); // filter
nums.reduce((a,b)=>a+b)  // sum

Objects

Key / value pairs — fundamental building block.

let person = {
  name: "Sita",
  age: 25,
  greet: function(){ return "Hi " + this.name }
};

// access
person.name;      // "Sita"
person['age'];    // 25
person.greet();   // "Hi Sita"
